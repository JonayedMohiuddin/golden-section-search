\begin{frame}{Four questions}
  \begin{enumerate}
    \item Why does the constant \(0.618\ldots\) appear?
    \item How can one function value be reused at every step?
    \item What accuracy is guaranteed after a fixed budget?
    \item When should we \emph{not} trust the answer?
  \end{enumerate}

  \begin{keyidea}[The working viewpoint]
    The method does not chase a derivative. It shrinks a bracket that is known
    to contain one optimum.
  \end{keyidea}
\end{frame}

\section{Problem Setting}

\begin{frame}{A derivative-free question}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\fontsize{22}{27}\selectfont
      \[
        x^*=\underset{x\in[x_L,x_U]}{\operatorname{arg\,max}}\; f(x)
      \]
      }
      \golden{Only request:} values of \(f(x)\).

      \vspace{0.20cm}
      A reliable derivative may be unavailable, expensive, or meaningless.

    \column{0.47\textwidth}
      \golden{Typical situations}
      \begin{itemize}
        \item laboratory measurements;
        \item simulation runs;
        \item noisy or tabulated output.
      \end{itemize}
      \muted{For minimization, replace \(\operatorname{arg\,max}\) by
      \(\operatorname{arg\,min}\); the geometry is unchanged.}
  \end{columns}
\end{frame}

\begin{frame}{The unimodality contract}
  \begin{columns}[c,onlytextwidth]
    \column{0.48\textwidth}
      \centering
      \input{figures/unimodal-maximum.tikz}

    \column{0.47\textwidth}
      A maximum at \(x^*\) divides the bracket:
      \begin{itemize}
        \item \(f\) is nondecreasing up to \(x^*\);
        \item \(f\) is nonincreasing after \(x^*\).
      \end{itemize}
      \begin{keyidea}[What is not required]
        No derivative, quadratic model, or smooth graph is required.
      \end{keyidea}
  \end{columns}
\end{frame}

\begin{frame}{The bracket is a claim}
  \begin{enumerate}
    \item \golden{Valid domain:} every trial point is safe to evaluate.
    \item \golden{Contains the target:} the desired optimum is inside.
    \item \golden{One relevant mode:} one peak or trough governs the interval.
  \end{enumerate}

  \vspace{0.08cm}
  \begin{keyidea}[Why this matters]
    The method certifies a smaller \emph{interval}; it cannot repair a bracket
    that excluded the optimum or mixed several extrema.
  \end{keyidea}
\end{frame}

\section{The Search Mechanism}

\begin{frame}{Two probes, one bracket}
  \centering
  \vspace{0.10cm}
  \input{figures/golden-probes.tikz}

  \vspace{0.45cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \[
        x_1=x_U-\tau L
      \]
    \column{0.46\textwidth}
      \[
        x_2=x_L+\tau L
      \]
  \end{columns}
  \vspace{-0.08cm}
  \[
    \tau=\frac{\sqrt5-1}{2}\approx0.618034,
    \qquad x_L<x_1<x_2<x_U.
  \]
\end{frame}

\begin{frame}{Which side survives?}
  \renewcommand{\arraystretch}{1.24}
  \centering
  \begin{tabular}{@{}lll@{}}
    \toprule
    Comparison & Maximization keeps & Minimization keeps \\
    \midrule
    \(f(x_1)>f(x_2)\) & \([x_L,x_2]\) & \([x_1,x_U]\) \\
    \(f(x_1)<f(x_2)\) & \([x_1,x_U]\) & \([x_L,x_2]\) \\
    \(f(x_1)=f(x_2)\) & \([x_1,x_2]\) & \([x_1,x_2]\) \\
    \bottomrule
  \end{tabular}

  \vspace{0.14cm}
  \begin{keyidea}[Read one row, not the whole table]
    For a maximum, the larger value points toward the survivor. For a minimum,
    the smaller value does.
  \end{keyidea}
\end{frame}

\begin{frame}{Reuse one evaluation}
  \centering
  \input{figures/point-reuse.tikz}

  \vspace{0.24cm}
  For the illustrated maximization case, \(f(x_1)>f(x_2)\). Therefore
  \[
    x_U'\gets x_2, \qquad x_2'\gets x_1,
    \qquad f(x_2')\gets f(x_1).
  \]
  Only \(x_1'\) is new, so only \golden{one new function call} is needed.
\end{frame}

\begin{frame}{Algorithm: initialize once}
  \begin{enumerate}
    \item Set \(\tau=(\sqrt5-1)/2\) and \(L=x_U-x_L\).
    \item Form \(x_1=x_U-\tau L\) and \(x_2=x_L+\tau L\).
    \item Evaluate \(f_1=f(x_1)\) and \(f_2=f(x_2)\).
  \end{enumerate}

  \vspace{0.18cm}
  \begin{keyidea}[State carried into the loop]
    Keep the ordered endpoints, two probes, and their two function values.
    Losing a saved value also loses the main efficiency advantage.
  \end{keyidea}
\end{frame}

\begin{frame}{Algorithm: repeat}
  \begin{enumerate}
    \item Compare \(f_1\) and \(f_2\).
    \item Discard the side ruled out by unimodality.
    \item Rename the surviving probe and reuse its value.
    \item Create one new probe and evaluate \(f\) there.
    \item Stop when the bracket meets the accuracy rule.
  \end{enumerate}

  \vspace{0.10cm}
  \muted{For minimization, reverse only the two strict comparison decisions;
  the probe geometry and reuse are unchanged.}
\end{frame}

\section{Convergence and Cost}

\begin{frame}{Why does 0.618 appear?}
  \begin{columns}[c,onlytextwidth]
    \column{0.56\textwidth}
      \centering
      \input{figures/golden-self-similarity.tikz}

    \column{0.41\textwidth}
      The old \(x_1\) is \((1-\tau)L\) from the left endpoint.

      In the new bracket, the right probe is \(\tau(\tau L)=\tau^2L\)
      from that endpoint. Reuse requires
      \[
        1-\tau=\tau^2.
      \]
      Hence
      \[
        \tau=\frac{\sqrt5-1}{2}.
      \]
  \end{columns}
\end{frame}

\begin{frame}{Why the discard is safe}
  Consider maximization with \(x_1<x_2\) and \(f(x_1)>f(x_2)\).

  \vspace{0.22cm}
  \begin{enumerate}
    \item Suppose, for contradiction, that \(x^*>x_2\).
    \item Both probes would then lie on the nondecreasing side of the peak.
    \item Unimodality would require \(f(x_1)\le f(x_2)\), a contradiction.
  \end{enumerate}

  \vspace{0.18cm}
  Therefore \(x^*\le x_2\), so discarding \((x_2,x_U]\) preserves the invariant
  \[
    \boxed{x^*\in[x_L,x_U]}.
  \]
  The other comparison and minimization cases are symmetric.
\end{frame}

\begin{frame}{Rate and error guarantee}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Exact contraction}
      \[
        L_k=\tau^kL_0.
      \]
      For midpoint output,
      \[
        |\widehat{x}_k-x^*|\le\frac{L_k}{2}.
      \]

    \column{0.49\textwidth}
      \golden{Required reductions}
      \[
        k\ge\max\left\{0,\left\lceil
        \frac{\log(2\varepsilon/L_0)}{\log\tau}
        \right\rceil\right\}.
      \]
      \begin{keyidea}[Example]
        If \(L_0=5\) and \(\varepsilon=10^{-3}\), then \(k=17\). The actual
        bound is about \(7.00\times10^{-4}\).
      \end{keyidea}
  \end{columns}
\end{frame}

\begin{frame}{What actually converges?}
  \golden{Bracket width:} exactly Q-linear,
  \[
    \frac{L_{k+1}}{L_k}=\tau.
  \]
  \golden{Location estimate:} its guaranteed error is at most \(L_k/2\), also
  decreasing linearly.

  \vspace{0.18cm}
  \golden{Objective value:} unimodality alone gives no numerical rate. Extra
  regularity is needed. For example, if \(f\) is \(M\)-Lipschitz,
  \[
    0\le f(x^*)-f(\widehat{x}_k)\le \frac{M L_k}{2}
  \]
  for maximization. A smooth stationary optimum can yield an
  \(O(L_k^2)\) objective error instead.
\end{frame}

\begin{frame}{Count calls, not loops}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Initialization}
      \[
        2\text{ function calls}
      \]
      produce the first comparison.

    \column{0.47\textwidth}
      \golden{Each later comparison}
      \[
        1\text{ new function call}
      \]
      because one value is reused.
  \end{columns}

  \vspace{0.22cm}
  With a stop-after-update implementation,
  \[
    n\text{ comparisons}\quad\Longrightarrow\quad n+1\text{ evaluations}.
  \]
  For an evaluation budget \(E\ge2\), the worst-case bracket is
  \[
    L_{E-1}=\tau^{E-1}L_0.
  \]
  \muted{This is why evaluation count is the meaningful complexity measure when
  \(f\) is a simulation or experiment.}
\end{frame}

\begin{frame}{Fixed-rule alternatives}
  \begin{enumerate}
    \item \golden{Equal-thirds search:} two new calls per comparison; retain
      \(2/3\) of the bracket.
    \item \golden{Fibonacci search:} finite-budget minimax optimal; changing
      ratios require choosing the budget first.
  \end{enumerate}

  \vspace{0.12cm}
  \golden{Golden section's compromise:} one stationary ratio, one reused value,
  and no predetermined final budget.
\end{frame}

\begin{frame}{Brent's hybrid}
  \begin{itemize}
    \item Propose an inverse parabolic interpolation step from sampled values.
    \item Accept it when safe; otherwise take a golden-section step.
  \end{itemize}

  \vspace{0.14cm}
  \golden{Trade-off:} golden section gives transparent worst-case contraction.
  Brent adds bookkeeping to gain speed when the curve is informative.
\end{frame}

\section{Reliable Computation}

\begin{frame}{Stop safely near zero}
  A common relative test contains \(x_U+x_L\) in the denominator. On a
  symmetric bracket such as \([-1,1]\), it divides by zero.

  \vspace{0.20cm}
  For midpoint output, use the scale-aware condition
  \[
    L_k\le 2\left(\varepsilon_{\rm abs}
      +\varepsilon_{\rm rel}\max\{|x_L|,|x_U|\}\right).
  \]
  It directly guarantees
  \[
    |\widehat{x}_k-x^*|\le
    \varepsilon_{\rm abs}+\varepsilon_{\rm rel}\max\{|x_L|,|x_U|\}.
  \]
  Also stop on an evaluation budget or when rounding creates no distinct probe.
\end{frame}

\begin{frame}{Implementation checks}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \raggedright
      \golden{Preserve state}
      \begin{itemize}
        \item move each \((x_i,f_i)\) pair together;
        \item cache the surviving evaluation;
        \item assert \(x_L<x_1<x_2<x_U\).
      \end{itemize}

    \column{0.47\textwidth}
      \raggedright
      \golden{Exit honestly}
      \begin{itemize}
        \item reject NaN or Inf values;
        \item stop if a probe rounds to an endpoint;
        \item return bracket, status, and call count.
      \end{itemize}
  \end{columns}

  \vspace{0.18cm}
  \muted{Once machine arithmetic cannot create a distinct probe, further
  interval algebra cannot create new information.}
\end{frame}

\begin{frame}{How the method misleads}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Several extrema}

      A legal-looking comparison can discard the globally best basin.

      \vspace{0.22cm}
      \golden{Noisy evaluations}

      One flipped comparison causes an irreversible wrong cut.

      \vspace{0.22cm}
      \golden{Flat optimum}

      The bracket shrinks, but the optimizer need not be unique.

    \column{0.47\textwidth}
      \golden{Invalid values}

      NaN, Inf, or domain errors destroy the ordering decision.

      \vspace{0.22cm}
      \golden{Wrong inequality}

      A minimization update used for maximization walks away from the peak.

      \vspace{0.22cm}
      \golden{False precision}

      A tiny bracket does not correct model error or measurement bias.
  \end{columns}
\end{frame}

\begin{frame}{Modern adaptations}
  \golden{Line search in many dimensions}
  \[
    \phi(\alpha)=F(\mathbf{x}+\alpha\mathbf{d})
  \]
  turns one step-length choice into a scalar golden-section problem.

  \golden{Noisy or expensive objectives}
  \begin{itemize}
    \item cache results, repeat uncertain comparisons, and enforce a budget;
    \item stop when noise exceeds the improvement being resolved.
  \end{itemize}

  \muted{The method is sequential: one new call per step saves evaluations, but
  the next location must wait. Parallel hardware may favor a batched alternative.}
\end{frame}

% Pure mathematical problem 1

\begin{frame}{Problem 1: Golden ratio}
  Normalize a search bracket to \([0,1]\). Place the two probes at
  \[
    x_1=1-\tau, \qquad x_2=\tau,
    \qquad \frac12<\tau<1.
  \]

  Suppose the right-hand bracket \([x_1,1]\) survives. Choose \(\tau\) so
  that the old point \(x_2\) is exactly the left probe of the new bracket.

  \begin{enumerate}
    \item Form the self-similarity equation for \(\tau\).
    \item Solve it and reject the inadmissible root.
    \item State the fraction of length removed by one reduction.
  \end{enumerate}

  \muted{No decimal approximation is needed until the last line.}
\end{frame}

\begin{frame}{Solution 1: Derivation}
  The surviving bracket has lower end \(1-\tau\) and length \(\tau\).
  Its left probe is therefore
  \[
    (1-\tau)+(1-\tau)\tau=1-\tau^2.
  \]
  Requiring this to coincide with the old point \(x_2=\tau\) gives
  \[
    1-\tau^2=\tau
    \quad\Longleftrightarrow\quad
    \tau^2+\tau-1=0.
  \]

  Hence
  \[
    \tau=\frac{-1\pm\sqrt5}{2}.
  \]
  The negative root is outside \((1/2,1)\), so
  \[
    \boxed{\tau=\frac{\sqrt5-1}{2}\approx0.618034}.
  \]

  The retained fraction is \(\tau\); the discarded fraction is
  \(1-\tau=\tau^2\approx0.381966\).
\end{frame}

\begin{frame}{Problem 2: Hand trace}
  Use golden-section search to minimize
  \[
    f(x)=(x-1.3)^2+2, \qquad 0\le x\le4.
  \]

  With \(\tau=(\sqrt5-1)/2\), carry out the first \textbf{four} bracket
  reductions. At each step, show
  \[
    x_1=x_U-\tau(x_U-x_L),
    \qquad
    x_2=x_L+\tau(x_U-x_L),
  \]
  the two function values, and the retained bracket.

  Finally, report the midpoint estimate and the number of function
  evaluations used.
\end{frame}

\begin{frame}{Solution 2: Trace table}
  For minimization, \(f(x_1)<f(x_2)\) keeps \([x_L,x_2]\); otherwise it keeps
  \([x_1,x_U]\).

  {\centering
  \fontsize{12.5}{15}\selectfont
  \setlength{\tabcolsep}{3.5pt}
  \begin{tabular}{c r r r r l}
    \toprule
    \(k\) & \(x_1\) & \(x_2\) & \(f(x_1)\) & \(f(x_2)\) & retained bracket \\
    \midrule
    1 & 1.5279 & 2.4721 & 2.0519 & 3.3739 & \([0,\,2.4721]\) \\
    2 & 0.9443 & 1.5279 & 2.1265 & 2.0519 & \([0.9443,\,2.4721]\) \\
    3 & 1.5279 & 1.8885 & 2.0519 & 2.3464 & \([0.9443,\,1.8885]\) \\
    4 & 1.3050 & 1.5279 & 2.0000 & 2.0519 & \([0.9443,\,1.5279]\) \\
    \bottomrule
  \end{tabular}\par}

  \vspace{0.18cm}
  Thus
  \[
    \widehat{x}=\frac{0.9443+1.5279}{2}=\boxed{1.2361}.
  \]
  Two calls initialize the method; each later reduction needs one new call.
  Four reductions therefore use \(2+3=\boxed{5}\) evaluations.

  \muted{Check: the exact minimizer \(1.3\) remains inside every retained bracket.}
\end{frame}

\begin{frame}{Problem 3: Error budget}
  A unimodal objective is bracketed on \([2,18]\). After \(k\) golden-section
  reductions, the midpoint of the surviving bracket will be reported.

  Find the smallest integer \(k\) that guarantees
  \[
    |\widehat{x}-x^*|\le 5\times10^{-4}.
  \]

  Your solution must:
  \begin{enumerate}
    \item start from the bracket-length formula;
    \item verify that one fewer reduction is insufficient;
    \item convert the result into a function-evaluation budget.
  \end{enumerate}
\end{frame}

\begin{frame}{Solution 3: Budget}
  The initial length is \(L_0=18-2=16\), and after \(k\) reductions
  \[
    L_k=16\tau^k,
    \qquad
    |\widehat{x}-x^*|\le\frac{L_k}{2}=8\tau^k.
  \]
  Therefore
  \[
    8\tau^k\le5\times10^{-4}
    \quad\Longrightarrow\quad
    k\ge
    \frac{\ln(6.25\times10^{-5})}{\ln\tau}
    \approx20.12.
  \]
  Thus the smallest integer is \(\boxed{k=21}\).

  The boundary check is
  \[
    8\tau^{20}\approx5.289\times10^{-4}>5\times10^{-4},
    \qquad
    8\tau^{21}\approx3.269\times10^{-4}<5\times10^{-4}.
  \]

  With cached reuse, \(k\) reductions require \(k+1\) calls. The guaranteed
  budget is therefore \(\boxed{22\text{ function evaluations}}\).
\end{frame}

% Pure mathematical problem 4
% -----------------------------------------------------------------------------
\begin{frame}{Problem 4: Safe discard}
  Let \(f\) be strictly unimodal with a unique maximum \(x^*\) on
  \([x_L,x_U]\), and let
  \[
    x_L<x_1<x_2<x_U.
  \]

  Prove directly from strict unimodality that
  \[
    f(x_1)<f(x_2)
    \quad\Longrightarrow\quad
    x^*\in[x_1,x_U].
  \]

  Then determine what the equality \(f(x_1)=f(x_2)\) says about the location
  of \(x^*\).

  \begin{keyidea}[Allowed fact]
    A strictly unimodal function increases before \(x^*\) and decreases after
    \(x^*\).
  \end{keyidea}
\end{frame}

\begin{frame}{Solution 4: Proof}
  \golden{Case 1: \(f(x_1)<f(x_2)\).}

  Suppose instead that \(x^*<x_1\). Both probes would then lie on the strictly
  decreasing side, so
  \[
    x^*<x_1<x_2
    \quad\Longrightarrow\quad
    f(x_1)>f(x_2),
  \]
  a contradiction. Therefore
  \[
    \boxed{x^*\in[x_1,x_U]}.
  \]

  \golden{Case 2: \(f(x_1)=f(x_2)\).}

  If \(x^*\le x_1\), strict decrease gives \(f(x_1)>f(x_2)\). If
  \(x^*\ge x_2\), strict increase gives \(f(x_1)<f(x_2)\). Both are impossible,
  hence
  \[
    \boxed{x^*\in(x_1,x_2)}.
  \]

  \muted{A flat-topped objective needs a weaker conclusion about its set of
  maximizers.}
\end{frame}

% -----------------------------------------------------------------------------
% Pure mathematical problem 5
% -----------------------------------------------------------------------------
\begin{frame}{Problem 5: Stopping rule}
  Consider the relative-width test
  \[
    2\left|\frac{x_U-x_L}{x_U+x_L}\right|<\delta.
  \]
  It is applied to \([x_L,x_U]=[-0.002,0.002]\) with \(\delta=10^{-3}\).

  \begin{enumerate}
    \item Explain exactly why the test fails.
    \item Replace it with a mixed absolute-relative test for the midpoint,
      using \(\varepsilon_{\rm abs}=10^{-4}\) and
      \(\varepsilon_{\rm rel}=10^{-3}\).
    \item Find how many additional golden reductions guarantee the new test.
  \end{enumerate}
\end{frame}

\begin{frame}{Solution 5: Scale floor}
  Here \(x_U+x_L=0\), so the original test is undefined even though the
  bracket length is \(0.004\). Use
  \[
    \frac{L}{2}
      \le \varepsilon_{\rm abs}
      +\varepsilon_{\rm rel}\max(|x_L|,|x_U|).
  \]
  The present length tolerance is \(2.04\times10^{-4}\). Since the relative
  scale can fall as the bracket moves, a uniform guarantee uses the absolute
  floor \(L\le2\times10^{-4}\). With \(L_m=0.004\tau^m\),
  \[
    m\ge
    \frac{\ln(2\times10^{-4}/0.004)}{\ln\tau}
    \approx6.23,
    \qquad \boxed{m=7}.
  \]

  Indeed, \(L_6\approx2.229\times10^{-4}\) fails, while
  \(L_7\approx1.378\times10^{-4}\) passes. The absolute term prevents the
  denominator pathology near zero.
\end{frame}

\begin{frame}{Application 1: Damper}
  A suspension engineer tunes a dimensionless damping setting \(c\). A
  finite-element run predicts the peak displacement
  \[
    D(c)=1.5+12(c-0.38)^2\ \text{mm},
    \qquad 0.10\le c\le0.90.
  \]
  The engineer treats the simulator as a black box and minimizes \(D\) using
  golden-section search. One run takes \(8\) minutes.

  \begin{enumerate}
    \item Carry out the first three bracket reductions.
    \item Report the final bracket and its midpoint.
    \item Find the total simulation time when values are cached.
  \end{enumerate}

  \muted{The formula is supplied so that the simulator outputs can be checked by hand.}
\end{frame}

\begin{frame}{Solution 1: Damper result}
  For minimization, retain the side containing the smaller displacement.

  \vspace{0.08cm}
  {\centering
  \fontsize{12.5}{15}\selectfont
  \setlength{\tabcolsep}{4.0pt}
  \begin{tabular}{c r r r r l}
    \toprule
    \(k\) & \(c_1\) & \(c_2\) & \(D(c_1)\) & \(D(c_2)\) & retained bracket \\
    \midrule
    1 & 0.4056 & 0.5944 & 1.5078 & 2.0517 & \([0.1000,\,0.5944]\) \\
    2 & 0.2889 & 0.4056 & 1.5997 & 1.5078 & \([0.2889,\,0.5944]\) \\
    3 & 0.4056 & 0.4777 & 1.5078 & 1.6146 & \([0.2889,\,0.4777]\) \\
    \bottomrule
  \end{tabular}\par}

  \vspace{0.16cm}
  Hence
  \[
    c\in[0.2889,0.4777],
    \qquad \widehat c=\frac{0.2889+0.4777}{2}=\boxed{0.3833}.
  \]
  Three reductions need \(3+1=4\) cached evaluations, so the elapsed simulation
  time is \(4(8)=\boxed{32\text{ minutes}}\).

  \muted{Audit: the model's exact best setting, \(c=0.38\), is inside the bracket.}
\end{frame}

\begin{frame}{Application 2: Insulation}
  A building model balances insulation cost against heat loss:
  \[
    C(t)=180t+100e^{-10t},
    \qquad 0.02\le t\le0.32,
  \]
  where \(t\) is thickness in metres and \(C\) is annual cost in thousands of
  taka. One energy-model run takes \(45\) seconds.

  Use golden-section search to guarantee a midpoint thickness error of at most
  \(1\) mm.

  \begin{enumerate}
    \item Determine the required reductions and model calls.
    \item Convert the call count into elapsed time.
    \item Report the computed bracket and midpoint thickness.
  \end{enumerate}
\end{frame}

\begin{frame}{Solution 2: Result}
  Here \(L_0=0.32-0.02=0.30\) m. The midpoint guarantee requires
  \[
    \frac{0.30\tau^k}{2}\le0.001
    \quad\Longrightarrow\quad
    k\ge\frac{\ln(0.001/0.15)}{\ln\tau}\approx10.41.
  \]
  Thus \(\boxed{k=11}\) reductions and \(\boxed{12}\) model calls are needed.
  The corresponding time is
  \[
    12(45)=540\text{ s}=\boxed{9\text{ minutes}}.
  \]

  Running the updates gives
  \[
    [t_L,t_U]=[0.170466,0.171973]\ \text{m},
    \qquad \widehat t=0.171220\ \text{m}=\boxed{171.2\text{ mm}}.
  \]

  \muted{Independent audit: \(C'(t)=0\) gives
  \(t^*=\ln(1000/180)/10=0.171480\) m, which lies in the bracket.}
\end{frame}

\begin{frame}{Application 3: Dosage}
  A dose-response surrogate assigns the net clinical benefit
  \[
    B(d)=95-0.12(d-14)^2,
    \qquad 5\le d\le25,
  \]
  where \(d\) is a dose in milligrams. The approved interval is unimodal, and
  the benefit must be maximized without using derivatives.

  The design requirement is stated in outcome units:
  \[
    B(d^*)-B(\widehat d)\le0.05\ \text{benefit points}.
  \]

  Find the smallest guaranteed reduction count, the evaluation count, and the
  midpoint recommendation produced by golden-section search.
\end{frame}

\begin{frame}{Solution 3: Benefit bound}
  After \(k\) reductions, the midpoint error obeys
  \[
    |\widehat d-d^*|\le10\tau^k
    \quad\Longrightarrow\quad
    B(d^*)-B(\widehat d)\le0.12(10\tau^k)^2.
  \]
  Requiring this bound to be at most \(0.05\) gives
  \[
    0.12(10\tau^k)^2\le0.05
    \quad\Longrightarrow\quad k\ge5.69.
  \]
  Therefore \(\boxed{k=6}\), requiring \(\boxed{7}\) benefit evaluations.

  The boundary check is \(0.0976>0.05\) at \(k=5\), but
  \(0.0373<0.05\) at \(k=6\).
  Six updates give \([13.3282,14.4427]\) mg, so
  \[
    \boxed{\widehat d=13.8854\text{ mg}}.
  \]
  \muted{Its actual model loss is only \(0.0016\); \(0.0373\) is the advance guarantee.}
\end{frame}

\begin{frame}{Application 4: Pricing}
  A retailer tests price \(p\), measured in hundreds of taka. A demand simulator
  predicts weekly profit, in thousands of taka, as
  \[
    \Pi(p)=12(p-8)e^{-0.06p},
    \qquad 10\le p\le40.
  \]
  Each simulator call takes \(20\) minutes.

  \begin{enumerate}
    \item Compute the initial golden probes and their profits.
    \item Perform the first maximization update and identify the reused point.
    \item For eight reductions, compare cached golden search with an
      implementation that evaluates two probes every time.
  \end{enumerate}
\end{frame}

\begin{frame}{Solution 4: Reuse pays}
  The initial probes and profits are
  \[
    p_1=21.4590,\quad \Pi(p_1)=44.568,
    \qquad
    p_2=28.5410,\quad \Pi(p_2)=44.472.
  \]
  Since \(\Pi(p_1)>\Pi(p_2)\), maximization retains
  \[
    \boxed{[10,28.5410]}.
  \]
  The old \(p_1=21.4590\) becomes the new right probe; only the new left probe
  \[
    p_1'=28.5410-\tau(28.5410-10)=17.0820
  \]
  needs evaluation.

  For eight reductions,
  \[
    \text{cached calls}=8+1=9,
    \qquad
    \text{two fresh probes each time}=16.
  \]
  Reuse saves \(7\) calls, or \(7(20)=\boxed{140\text{ minutes}}\).
\end{frame}

\begin{frame}{Application 5: Solar tilt}
  A solar-field simulation maximizes daily energy over
  \(10^\circ\le\theta\le50^\circ\). The first golden probes are
  \[
    \theta_1=25.279^\circ,
    \qquad \theta_2=34.721^\circ.
  \]
  Weather variability produces these 95\% intervals, in kWh:
  \[
    \theta_1: 5.84\pm0.06,
    \qquad
    \theta_2: 5.88\pm0.06.
  \]
  After replication they narrow to \([5.82,5.86]\) and \([5.87,5.91]\).

  Explain the correct decision before and after replication, find the next new
  probe, and calculate the cost of five runs per probe at two minutes per run.
\end{frame}

\begin{frame}{Solution 5}
  Initially the intervals overlap, so the sign of
  \(E(\theta_2)-E(\theta_1)\) is unresolved. A standard golden-section discard
  would risk deleting the true maximum.

  After replication,
  \[
    [5.82,5.86]\cap[5.87,5.91]=\varnothing,
  \]
  and the second probe is reliably better. For maximization, retain
  \[
    \boxed{[25.279^\circ,50^\circ]}.
  \]
  The old \(34.721^\circ\) point becomes the new left probe. The new right probe is
  \[
    25.279+\tau(50-25.279)=\boxed{40.557^\circ}.
  \]

  Five runs at each original probe require \(10\) runs, costing
  \(10(2)=\boxed{20\text{ minutes}}\). Later comparisons should also replicate
  only when their uncertainty intervals overlap.
\end{frame}

\section{Summary}
% Final checklist and takeaways.

\begin{frame}{Core result}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Geometry}
      \[
        \tau=\frac{\sqrt5-1}{2},
        \qquad
        L_k=\tau^kL_0.
      \]
      If the midpoint is returned,
      \[
        |\widehat{x}_k-x^*|\le\frac{L_k}{2}.
      \]
      Each comparison retains \(61.8\%\) and discards \(38.2\%\) of the
      current bracket.

    \column{0.47\textwidth}
      \golden{Computation}
      \begin{itemize}
        \item two calls initialize the search;
        \item one old value is reused at every later reduction;
        \item \(k\) reductions cost \(k+1\) function evaluations.
      \end{itemize}
  \end{columns}

  \vspace{0.12cm}
  \muted{The bracket converges Q-linearly. Objective-value accuracy needs
  additional information about the local shape of \(f\).}
\end{frame}

\begin{frame}{Use it deliberately}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Use it when}
      \begin{itemize}\setlength{\itemsep}{0.05cm}
        \item there is one scalar variable;
        \item the bracket is plausibly unimodal;
        \item evaluations are costly and derivatives are unreliable.
      \end{itemize}

    \column{0.47\textwidth}
      \golden{Switch when}
      \begin{itemize}\setlength{\itemsep}{0.05cm}
        \item interpolation is trustworthy: use Brent's hybrid;
        \item several extrema require global or multistart search;
        \item noise or vector decisions require a different solver.
      \end{itemize}
  \end{columns}

  \vspace{0.08cm}
  \muted{Report the bracket, tolerance, call count, and exit status.}
\end{frame}
