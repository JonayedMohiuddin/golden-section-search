\begin{frame}{Four questions}
  \begin{enumerate}
    \item Why does the constant \(0.618\ldots\) appear?
    \item How can one function value be reused at every step?
    \item What accuracy is guaranteed after a fixed budget?
    \item When should we \emph{not} trust the answer?
  \end{enumerate}

  \vspace{0.05cm}
  \begin{keyidea}[The working viewpoint]
    The method does not chase a derivative. It shrinks a bracket that is known
    to contain one optimum.
  \end{keyidea}
\end{frame}

\section{Problem Setting}

\begin{frame}{A derivative-free question}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\fontsize{22}{27}\selectfont
      \[
        x^*=\underset{x\in[x_L,x_U]}{\operatorname{arg\,max}}\; f(x)
      \]
      }
      \golden{Only request:} values of \(f(x)\).

      \vspace{0.20cm}
      A reliable derivative may be unavailable, expensive, or meaningless.

    \column{0.47\textwidth}
      \golden{Typical situations}
      \begin{itemize}
        \item laboratory measurements;
        \item simulation runs;
        \item noisy or tabulated output.
      \end{itemize}
      \muted{For minimization, replace \(\operatorname{arg\,max}\) by
      \(\operatorname{arg\,min}\); the geometry is unchanged.}
  \end{columns}
\end{frame}

\begin{frame}{The unimodality contract}
  \begin{columns}[c,onlytextwidth]
    \column{0.48\textwidth}
      \centering
      \input{figures/unimodal-maximum.tikz}

    \column{0.47\textwidth}
      A maximum at \(x^*\) divides the bracket:
      \begin{itemize}
        \item \(f\) is nondecreasing up to \(x^*\);
        \item \(f\) is nonincreasing after \(x^*\).
      \end{itemize}
      \begin{keyidea}[What is not required]
        No derivative, quadratic model, or smooth graph is required.
      \end{keyidea}
  \end{columns}
  \sourceline{Course notes, p.~1; maximization convention retained.}
\end{frame}

\begin{frame}{The bracket is a claim}
  \begin{enumerate}
    \item \golden{Valid domain:} every trial point is safe to evaluate.
    \item \golden{Contains the target:} the desired optimum is inside.
    \item \golden{One relevant mode:} one peak or trough governs the interval.
  \end{enumerate}

  \vspace{0.08cm}
  \begin{keyidea}[Why this matters]
    The method certifies a smaller \emph{interval}; it cannot repair a bracket
    that excluded the optimum or mixed several extrema.
  \end{keyidea}
\end{frame}

\section{The Search Mechanism}

\begin{frame}{Two probes, one bracket}
  \centering
  \vspace{0.10cm}
  \input{figures/golden-probes.tikz}

  \vspace{0.45cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \[
        x_1=x_U-\tau L
      \]
    \column{0.46\textwidth}
      \[
        x_2=x_L+\tau L
      \]
  \end{columns}
  \vspace{-0.08cm}
  \[
    \tau=\frac{\sqrt5-1}{2}\approx0.618034,
    \qquad x_L<x_1<x_2<x_U.
  \]
  \sourceline{Course notes, pp.~1-2. The derivation of \(\tau\) follows in the analysis section.}
\end{frame}

\begin{frame}{Which side survives?}
  \renewcommand{\arraystretch}{1.24}
  \centering
  \begin{tabular}{@{}lll@{}}
    \toprule
    Comparison & Maximization keeps & Minimization keeps \\
    \midrule
    \(f(x_1)>f(x_2)\) & \([x_L,x_2]\) & \([x_1,x_U]\) \\
    \(f(x_1)<f(x_2)\) & \([x_1,x_U]\) & \([x_L,x_2]\) \\
    \(f(x_1)=f(x_2)\) & \([x_1,x_2]\) & \([x_1,x_2]\) \\
    \bottomrule
  \end{tabular}

  \vspace{0.14cm}
  \begin{keyidea}[Read one row, not the whole table]
    For a maximum, the larger value points toward the survivor. For a minimum,
    the smaller value does.
  \end{keyidea}
  \sourceline{The strict-inequality cases follow the update rule in the course notes, pp.~2-3.}
\end{frame}

\begin{frame}{Reuse one evaluation}
  \centering
  \input{figures/point-reuse.tikz}

  \vspace{0.24cm}
  For the illustrated maximization case, \(f(x_1)>f(x_2)\). Therefore
  \[
    x_U'\gets x_2, \qquad x_2'\gets x_1,
    \qquad f(x_2')\gets f(x_1).
  \]
  Only \(x_1'\) is new, so only \golden{one new function call} is needed.
  \sourceline{Course notes, p.~2.}
\end{frame}

\begin{frame}{Algorithm: initialize once}
  \begin{enumerate}
    \item Set \(\tau=(\sqrt5-1)/2\) and \(L=x_U-x_L\).
    \item Form \(x_1=x_U-\tau L\) and \(x_2=x_L+\tau L\).
    \item Evaluate \(f_1=f(x_1)\) and \(f_2=f(x_2)\).
  \end{enumerate}

  \vspace{0.18cm}
  \begin{keyidea}[State carried into the loop]
    Keep the ordered endpoints, two probes, and their two function values.
    Losing a saved value also loses the main efficiency advantage.
  \end{keyidea}
\end{frame}

\begin{frame}{Algorithm: repeat}
  \begin{enumerate}
    \item Compare \(f_1\) and \(f_2\).
    \item Discard the side ruled out by unimodality.
    \item Rename the surviving probe and reuse its value.
    \item Create one new probe and evaluate \(f\) there.
    \item Stop when the bracket meets the accuracy rule.
  \end{enumerate}

  \vspace{0.10cm}
  \muted{For minimization, reverse only the two strict comparison decisions;
  the probe geometry and reuse are unchanged.}
\end{frame}

\section{Convergence and Cost}

\begin{frame}{Why does 0.618 appear?}
  \begin{columns}[c,onlytextwidth]
    \column{0.56\textwidth}
      \centering
      \input{figures/golden-self-similarity.tikz}

    \column{0.41\textwidth}
      The old \(x_1\) is \((1-\tau)L\) from the left endpoint.

      In the new bracket, the right probe is \(\tau(\tau L)=\tau^2L\)
      from that endpoint. Reuse requires
      \[
        1-\tau=\tau^2.
      \]
      Hence
      \[
        \tau=\frac{\sqrt5-1}{2}.
      \]
  \end{columns}
  \sourceline{Course notes, pp.~4-5; Kiefer (1953) formalized finite-budget sequential minimax search.}
\end{frame}

\begin{frame}{Why the discard is safe}
  Consider maximization with \(x_1<x_2\) and \(f(x_1)>f(x_2)\).

  \vspace{0.22cm}
  \begin{enumerate}
    \item Suppose, for contradiction, that \(x^*>x_2\).
    \item Both probes would then lie on the nondecreasing side of the peak.
    \item Unimodality would require \(f(x_1)\le f(x_2)\), a contradiction.
  \end{enumerate}

  \vspace{0.18cm}
  Therefore \(x^*\le x_2\), so discarding \((x_2,x_U]\) preserves the invariant
  \[
    \boxed{x^*\in[x_L,x_U]}.
  \]
  The other comparison and minimization cases are symmetric.
\end{frame}

\begin{frame}{Rate and error guarantee}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Exact contraction}
      \[
        L_k=\tau^kL_0.
      \]
      For midpoint output,
      \[
        |\widehat{x}_k-x^*|\le\frac{L_k}{2}.
      \]

    \column{0.49\textwidth}
      \golden{Required reductions}
      \[
        k\ge\max\left\{0,\left\lceil
        \frac{\log(2\varepsilon/L_0)}{\log\tau}
        \right\rceil\right\}.
      \]
      \begin{keyidea}[Example]
        If \(L_0=5\) and \(\varepsilon=10^{-3}\), then \(k=17\). The actual
        bound is about \(7.00\times10^{-4}\).
      \end{keyidea}
  \end{columns}
\end{frame}

\begin{frame}{What actually converges?}
  \golden{Bracket width:} exactly Q-linear,
  \[
    \frac{L_{k+1}}{L_k}=\tau.
  \]
  \golden{Location estimate:} its guaranteed error is at most \(L_k/2\), also
  decreasing linearly.

  \vspace{0.18cm}
  \golden{Objective value:} unimodality alone gives no numerical rate. Extra
  regularity is needed. For example, if \(f\) is \(M\)-Lipschitz,
  \[
    0\le f(x^*)-f(\widehat{x}_k)\le \frac{M L_k}{2}
  \]
  for maximization. A smooth stationary optimum can yield an
  \(O(L_k^2)\) objective error instead.
\end{frame}

\begin{frame}{Count calls, not loops}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Initialization}
      \[
        2\text{ function calls}
      \]
      produce the first comparison.

    \column{0.47\textwidth}
      \golden{Each later comparison}
      \[
        1\text{ new function call}
      \]
      because one value is reused.
  \end{columns}

  \vspace{0.22cm}
  With a stop-after-update implementation,
  \[
    n\text{ comparisons}\quad\Longrightarrow\quad n+1\text{ evaluations}.
  \]
  For an evaluation budget \(E\ge2\), the worst-case bracket is
  \[
    L_{E-1}=\tau^{E-1}L_0.
  \]
  \muted{This is why evaluation count is the meaningful complexity measure when
  \(f\) is a simulation or experiment.}
\end{frame}

\begin{frame}{Fixed-rule alternatives}
  \begin{enumerate}
    \item \golden{Equal-thirds search:} two new calls per comparison; retain
      \(2/3\) of the bracket.
    \item \golden{Fibonacci search:} finite-budget minimax optimal; changing
      ratios require choosing the budget first.
  \end{enumerate}

  \vspace{0.12cm}
  \golden{Golden section's compromise:} one stationary ratio, one reused value,
  and no predetermined final budget.
  \sourceline{Kiefer, ``Sequential minimax search for a maximum,'' Proc. AMS 4 (1953), 502-506.}
\end{frame}

\begin{frame}{Brent's hybrid}
  \begin{itemize}
    \item Propose an inverse parabolic interpolation step from sampled values.
    \item Accept it when safe; otherwise take a golden-section step.
  \end{itemize}

  \vspace{0.14cm}
  \golden{Trade-off:} golden section gives transparent worst-case contraction.
  Brent adds bookkeeping to gain speed when the curve is informative.
  \sourceline{SciPy 1.18 \texttt{minimize\_scalar} documentation; Brent (1973).}
\end{frame}

\section{Reliable Computation}

\begin{frame}{Stop safely near zero}
  The course-note relative test contains \(x_U+x_L\) in the denominator. On a
  symmetric bracket such as \([-1,1]\), it divides by zero.

  \vspace{0.20cm}
  For midpoint output, use the scale-aware condition
  \[
    L_k\le 2\left(\varepsilon_{\rm abs}
      +\varepsilon_{\rm rel}\max\{|x_L|,|x_U|\}\right).
  \]
  It directly guarantees
  \[
    |\widehat{x}_k-x^*|\le
    \varepsilon_{\rm abs}+\varepsilon_{\rm rel}\max\{|x_L|,|x_U|\}.
  \]
  Also stop on an evaluation budget or when rounding creates no distinct probe.
  \sourceline{Course notes, p.~3; compare SciPy's absolute \texttt{xatol} and \texttt{maxfun} controls.}
\end{frame}

\begin{frame}{Implementation guardrails}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Preserve state}
      \begin{itemize}
        \item move each \((x_i,f_i)\) pair together;
        \item cache the surviving evaluation;
        \item assert \(x_L<x_1<x_2<x_U\).
      \end{itemize}

    \column{0.47\textwidth}
      \golden{Exit honestly}
      \begin{itemize}
        \item reject NaN or Inf values;
        \item stop if a probe rounds to an endpoint;
        \item return bracket, status, and call count.
      \end{itemize}
  \end{columns}

  \vspace{0.18cm}
  \muted{Once machine arithmetic cannot create a distinct probe, further
  interval algebra cannot create new information.}
\end{frame}

\begin{frame}{How the method misleads}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \golden{Several extrema}

      A legal-looking comparison can discard the globally best basin.

      \vspace{0.22cm}
      \golden{Noisy evaluations}

      One flipped comparison causes an irreversible wrong cut.

      \vspace{0.22cm}
      \golden{Flat optimum}

      The bracket shrinks, but the optimizer need not be unique.

    \column{0.47\textwidth}
      \golden{Invalid values}

      NaN, Inf, or domain errors destroy the ordering decision.

      \vspace{0.22cm}
      \golden{Wrong inequality}

      A minimization update used for maximization walks away from the peak.

      \vspace{0.22cm}
      \golden{False precision}

      A tiny bracket does not correct model error or measurement bias.
  \end{columns}
\end{frame}

\begin{frame}{Modern adaptations}
  \golden{Line search in many dimensions}
  \[
    \phi(\alpha)=F(\mathbf{x}+\alpha\mathbf{d})
  \]
  turns one step-length choice into a scalar golden-section problem.

  \golden{Noisy or expensive objectives}
  \begin{itemize}
    \item cache results, repeat uncertain comparisons, and enforce a budget;
    \item stop when noise exceeds the improvement being resolved.
  \end{itemize}

  \muted{The method is sequential: one new call per step saves evaluations, but
  the next location must wait. Parallel hardware may favor a batched alternative.}
\end{frame}
